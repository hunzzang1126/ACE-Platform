export function classifyRatio(w: number, h: number): SizeCategory {
    const ratio = w / h;
    if (ratio > 4.0) return 'ultra-wide';
    if (ratio > 2.0) return 'wide';
    if (ratio > 1.2) return 'landscape';

    if (ratio > 0.5) return 'portrait';
    return 'ultra-tall';
}
